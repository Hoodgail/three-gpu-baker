cmake_minimum_required(VERSION 3.20)
project(lightmap_optix LANGUAGES CXX)
set(CMAKE_CXX_STANDARD 17)
set(CMAKE_CXX_STANDARD_REQUIRED ON)
find_package(CUDAToolkit REQUIRED)
set(OPTIX_ROOT "" CACHE PATH "OptiX SDK root containing include/optix.h")
find_path(OPTIX_INCLUDE_DIR optix.h HINTS "${OPTIX_ROOT}/include" "$ENV{OPTIX_ROOT}/include" REQUIRED)
add_executable(lightmap-optix main.cpp)
target_include_directories(lightmap-optix PRIVATE "${OPTIX_INCLUDE_DIR}")
target_link_libraries(lightmap-optix PRIVATE CUDA::cudart CUDA::cuda_driver ${CMAKE_DL_LIBS})
